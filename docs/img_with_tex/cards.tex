\newcommand{\drawcard}[1]{%
  \begin{tikzpicture}
  \draw[cardbg] (0,0) rectangle (\cardwidth,\cardheight);
  \draw[innerborder] (0.3,0.3) rectangle (\cardwidth-0.3,\cardheight-0.3);
  \coordinate (Center) at (\cardwidth/2, \cardheight/2);
  \draw[gray!40, line width=0.6pt] (Center) circle (1.8cm);
  \draw[gray!40, line width=0.6pt] (Center) circle (2.4cm);
  \foreach \angle in {0, 30, ..., 330} { \draw[gray!40, line width=0.6pt] (Center) ++(\angle:1.4cm) circle (1.4cm); }
  \node[anchor=north west, font=\Huge\bfseries, color=black] at (0.4, \cardheight-0.4) {\ensuremath{#1}};
  \node[rotate=180, anchor=north west, font=\Huge\bfseries, color=black] at (\cardwidth-0.4, 0.4) {\ensuremath{#1}};
  \node[font=\fontsize{50}{0}, color=black, anchor=center] at (\cardwidth/2, \cardheight*0.65) {\ensuremath{#1}};
  \node[font=\fontsize{50}{0}, rotate=180, color=black, anchor=center] at (\cardwidth/2, \cardheight*0.35) {\ensuremath{#1}};
  \end{tikzpicture}%
}
